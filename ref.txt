import json
from typing import Any, Dict, Tuple, Type
from pydantic import create_model

# Map string type names to Python types
type_map = {
    "str": str,
    "int": int,
    "float": float,
    "bool": bool
}

def create_dynamic_model_from_json(model_name: str, json_str: str) -> Type:
    """
    Create a Pydantic model dynamically from JSON string of fields.

    Args:
        model_name (str): Name of the model.
        json_str (str): JSON string like '{"temperature": "float", "condition": "str"}'

    Returns:
        Type: A dynamically created Pydantic model.
    """
    fields_dict: Dict[str, str] = json.loads(json_str)
    fields: Dict[str, Tuple[Any, Any]] = {}

    for field_name, type_str in fields_dict.items():
        py_type = type_map.get(type_str.strip())
        if not py_type:
            raise ValueError(f"Unsupported type: {type_str}")
        fields[field_name] = (py_type, ...)

    return create_model(model_name, **fields)
json_input = '{"temperature": "float", "condition": "str"}'
DynamicWeatherModel = create_dynamic_model_from_json("WeatherModel", json_input)

instance = DynamicWeatherModel(temperature=25.0, condition="Rainy")
print(instance)
